WebSocket — Deep Internals

Protocol handshake, STOMP, SockJS, scaling, sticky sessions, real-time chat

1. WebSocket Handshake Protocol

Step 1: Client sends HTTP Upgrade request GET /ws HTTP/1.1 Host: chat.example.com Connection: Upgrade Upgrade: websocket Sec-WebSocket-Key: dGhlIHNhbXBsZSBub25jZQ== ← random base64 Sec-WebSocket-Version: 13 Sec-WebSocket-Protocol: chat, superchat ← requested sub-protocols Origin: https://app.example.com Step 2: Server responds 101 Switching Protocols HTTP/1.1 101 Switching Protocols Connection: Upgrade Upgrade: websocket Sec-WebSocket-Accept: s3pPLMBiTxaQ9kYGzzhZRbK+xOo= ← SHA1(key + GUID) Sec-WebSocket-Protocol: chat ← chosen sub-protocol Step 3: Protocol switch TCP connection remains open HTTP framing discarded WebSocket frames replace HTTP: {FIN, RSV, opcode, MASK, payload_len, payload} Full duplex: both sides can send anytime, no request/response cycle

2. WebSocket Frame Structure

// WebSocket data frame (binary format):
// Bit: 0       4       8       16               32+
//      FIN|RSV|opcode|MASK|payload_len   |extended_len|mask_key|payload
//
// Opcodes:
// 0x0 = continuation frame
// 0x1 = text frame (UTF-8)
// 0x2 = binary frame
// 0x8 = close  (close code + reason)
// 0x9 = ping   (client or server sends to check alive)
// 0xA = pong   (response to ping)
//
// Heartbeat mechanism:
// Server pings every 30s → client pongs → server knows client alive
// No pong within timeout → server closes connection
spring.websocket.ping-interval=30000    # ms between pings
spring.websocket.ping-timeout=5000     # ms to wait for pong

3. STOMP Over WebSocket

STOMP = Simple Text Oriented Messaging Protocol Adds pub/sub semantics ON TOP of WebSocket frames Client subscribes to destinations: /topic/chat-room-1 Server broadcasts to that destination Enables: @MessageMapping, @SendTo, /topic, /queue STOMP Frame format: SUBSCRIBE id:sub-0 destination:/topic/room.123 ^@ SEND destination:/app/chat.send content-type:application/json content-length:45 {"roomId":"123","message":"Hello everyone!"}^@ SERVER → CLIENT (broadcast): MESSAGE destination:/topic/room.123 message-id:msg-001 subscription:sub-0 {"from":"Alice","message":"Hello everyone!","time":"..."}^@

                .setHttpMessageCacheSize(1000)

        reg.setMessageSizeLimit(128 * 1024);  // 128KB max message
        reg.setSendBufferSizeLimit(512 * 1024);
        reg.setSendTimeLimit(20 * 1000);       // 20s send timeout
    }

@Controller
public class ChatController {

    private final SimpMessagingTemplate messagingTemplate;
    private final ChatMessageService chatService;
    private final RedisTemplate<String, Object> redis;

    // Route: /app/chat.send → broadcast to /topic/room.{roomId}
    @MessageMapping("/chat.send/{roomId}")
    @SendTo("/topic/room.{roomId}")
    public ChatMessageDto sendMessage(
            @DestinationVariable String roomId,
            @Payload SendMessageRequest request,
            Principal principal) {

        ChatMessage saved = chatService.save(roomId, principal.getName(), request.getText());

        // Track online users in Redis
        redis.opsForSet().add("room:" + roomId + ":members", principal.getName());

        return ChatMessageDto.from(saved);
        // @SendTo broadcasts return value to /topic/room.{roomId}
    }

    // Private message: /app/chat.private → /user/{targetUser}/queue/messages
    @MessageMapping("/chat.private")
    public void privateMessage(@Payload PrivateMessageRequest req, Principal sender) {
        ChatMessageDto msg = ChatMessageDto.builder()
            .from(sender.getName()).to(req.getTo())
            .text(req.getText()).timestamp(Instant.now()).build();

        // Send only to specific user (all their sessions)
        messagingTemplate.convertAndSendToUser(req.getTo(), "/queue/messages", msg);
        messagingTemplate.convertAndSendToUser(sender.getName(), "/queue/messages", msg);
    }

    // Typing indicator — no DB persistence needed
    @MessageMapping("/chat.typing/{roomId}")
    @SendTo("/topic/room.{roomId}.typing")
    public TypingEvent typing(@DestinationVariable String roomId, Principal user) {
        return new TypingEvent(user.getName(), Instant.now());
    }

    // Push notification from backend (not triggered by message)
    public void pushSystemMessage(String roomId, String text) {
        messagingTemplate.convertAndSend("/topic/room." + roomId,
            ChatMessageDto.system(text));
    }
}

            .simpDestMatchers("/app/**").authenticated()
            .simpSubscribeDestMatchers("/topic/room.*").hasRole("USER")
            .anyMessage().authenticated();

    @Override

// Channel interceptor: validate JWT on CONNECT frame
@Component
public class WebSocketAuthInterceptor implements ChannelInterceptor {
    @Override
    public Message<?> preSend(Message<?> message, MessageChannel channel) {
        StompHeaderAccessor accessor = MessageHeaderAccessor.getAccessor(message, StompHeaderAccessor.class);
        if (StompCommand.CONNECT.equals(accessor.getCommand())) {
            String token = accessor.getFirstNativeHeader("Authorization");
            if (token != null && token.startsWith("Bearer ")) {
                UsernamePasswordAuthenticationToken auth = jwtService.validate(token.substring(7));
                accessor.setUser(auth);
            } else { throw new AccessDeniedException("Missing auth"); }
        }
        return message;
    }
}

5. Scaling WebSocket — The Hard Problem

Problem: WebSocket connections are stateful (persistent TCP) User A connected to Server-1 User B connected to Server-2 A sends message to room → Server-1 broadcast → only A's server-1-connected peers see it! B never receives the message! Solution 1: Sticky Sessions (simple but fragile) Load balancer routes user to SAME server on reconnect Nginx: ip_hash; (same IP → same server) AWS ALB: enable "Stickiness" (cookie-based) Problem: server crash → all its sessions lost, users must reconnect Solution 2: External Message Broker (Redis Pub/Sub or RabbitMQ) Server-1 receives message → publishes to Redis channel "room:123" Redis broadcasts to ALL servers subscribed to "room:123" Server-2 receives from Redis → delivers to its connected users → No sticky sessions needed! Any server handles any user Implementation: Spring WebSocket + STOMP + RabbitMQ broker relay config.enableStompBrokerRelay("/topic", "/queue") .setRelayHost("rabbitmq") .setRelayPort(61613) .setClientLogin("guest").setClientPasscode("guest"); Solution 3: Dedicated WebSocket Gateway (Socket.IO cluster, Centrifugo) Specialized service for connection management Backend services publish to broker Gateway fan-out to connected clients

Kafka Streams

Stream processing, windowing, aggregations, joins, fault tolerance, ETL pipelines

1. Kafka Streams Architecture

Kafka Streams runs INSIDE your application (no separate cluster!) Input Topics → Stream Topology → Output Topics │ ├── Source Processors (read from Kafka topic) ├── Stream Processors (transform, filter, aggregate) └── Sink Processors (write to Kafka topic) State Stores: Local RocksDB (on disk) per task instance Changelog topic in Kafka (backup/recovery) On restart: replay changelog → restore state → Fault-tolerant stateful processing! Partitioning = Parallelism: Topic: orders (6 partitions) App instance with 2 threads: processes 3 partitions each Scale to 6 instances: 1 partition per instance (max parallelism) Each partition processed in order → per-key ordering guaranteed

2. Core DSL Operations

@Bean
public Topology buildTopology() {
    StreamsBuilder builder = new StreamsBuilder();

    // Source stream from topic
    KStream<String, Order> orders = builder.stream("orders",
        Consumed.with(Serdes.String(), orderSerde));

    // Filter: only COMPLETED orders
    KStream<String, Order> completed = orders
        .filter((key, order) -> order.getStatus() == COMPLETED);

    // Map: transform to revenue record
    KStream<String, Revenue> revenues = completed
        .mapValues(order -> new Revenue(order.getCustomerId(), order.getAmount()))
        .selectKey((key, rev) -> rev.getCustomerId()); // repartition by customerId

    // Branching: split stream
    Map<String, KStream<String, Order>> branches = orders
        .split(Named.as("branch-"))
        .branch((k, o) -> o.getAmount().compareTo(BigDecimal.valueOf(100)) > 0,
                Branched.as("large"))
        .defaultBranch(Branched.as("small"));

    branches.get("branch-large").to("large-orders");
    branches.get("branch-small").to("small-orders");

    // Sink to topic
    revenues.to("customer-revenues", Produced.with(Serdes.String(), revenueSerde));

    return builder.build();
}

// |--5min--|--5min--|--5min--|
KTable<Windowed<String>, Long> orderCountPerWindow = orders
    .groupByKey()
    .windowedBy(TimeWindows.ofSizeWithNoGrace(Duration.ofMinutes(5)))
    .count(Materialized.as("order-count-store"));

orderCountPerWindow.toStream()
    .map((windowedKey, count) -> KeyValue.pair(
        windowedKey.key() + "@" + windowedKey.window().startTime(),
        new WindowedCount(windowedKey.key(), windowedKey.window().startTime(), count)))
    .to("windowed-counts");

// Sliding window: overlapping, captures last N time units
// Used for: running metrics, anomaly detection
KTable<Windowed<String>, Double> avgRevenue = orders
    .groupBy((k, o) -> KeyValue.pair(o.getCategory(), o.getAmount()),
             Grouped.with(Serdes.String(), bigDecimalSerde))
    .windowedBy(SlidingWindows.ofTimeDifferenceWithNoGrace(Duration.ofMinutes(10)))
    .aggregate(
        () -> new RevenueAccumulator(0L, BigDecimal.ZERO),
        (key, order, acc) -> acc.add(order),
        Materialized.as("revenue-store")
    );

// Session window: activity-based, gap triggers new window
// Useful for: user session analytics
orders.groupByKey()
    .windowedBy(SessionWindows.ofInactivityGapWithNoGrace(Duration.ofMinutes(30)))
    .count()  // new session if 30min gap in user activity
    .toStream().to("user-sessions");

5. ETL Pipeline — Real-Time Data Processing

// ETL: raw events → validated → enriched → aggregated → sink
@Bean
public Topology orderAnalyticsPipeline() {
    StreamsBuilder builder = new StreamsBuilder();

    // E: Extract
    KStream<String, RawEvent> raw = builder.stream("raw-events");

    // T: Transform — validate, filter, enrich
    KStream<String, ProcessedEvent> processed = raw
        .filter((k, e) -> e.getPayload() != null)
        .mapValues(this::validate)
        .filter((k, e) -> e.isValid())
        .mapValues(this::enrich)
        .peek((k, e) -> metrics.counter("events.processed").increment());

    // Aggregate: count per event type per hour
    KTable<Windowed<String>, Long> hourlyStats = processed
        .selectKey((k, e) -> e.getEventType())
        .groupByKey()
        .windowedBy(TimeWindows.ofSizeWithNoGrace(Duration.ofHours(1)))
        .count(Materialized.as("hourly-event-counts"));

    // L: Load — write to output topics
    processed.to("processed-events");
    hourlyStats.toStream()
        .mapValues((wk, count) -> new HourlyStat(wk.key(), wk.window().start(), count))
        .selectKey((wk, stat) -> stat.getEventType())
        .to("hourly-stats");

    return builder.build();
}

@RestController
public class StreamQueryController {
    private final InteractiveQueryService queryService;

    @GetMapping("/analytics/order-count/{customerId}")
    public Long getOrderCount(@PathVariable String customerId) {
        ReadOnlyKeyValueStore<String, Long> store =
            queryService.getQueryableStore("order-count-store",
                QueryableStoreTypes.keyValueStore());
        return Optional.ofNullable(store.get(customerId)).orElse(0L);
    }

    // For windowed stores:
    @GetMapping("/analytics/windowed/{key}")
    public List<WindowedCount> getWindowed(@PathVariable String key) {
        ReadOnlyWindowStore<String, Long> store =
            queryService.getQueryableStore("order-count-store",
                QueryableStoreTypes.windowStore());
        WindowStoreIterator<Long> iter = store.fetch(key,
            Instant.now().minus(Duration.ofHours(1)), Instant.now());
        return StreamSupport.stream(
            Spliterators.spliteratorUnknownSize(iter, 0), false)
            .map(kv -> new WindowedCount(key, kv.key, kv.value)).toList();
    }
}

1. Docker Architecture

Docker Architecture: dockerd (daemon) ← API server, manages containers/images containerd ← container runtime (lifecycle management) runc ← OCI runtime (actually creates Linux namespaces/cgroups) Linux kernel features powering containers: Namespaces: PID, Network, Mount, UTS, IPC, User → isolation cgroups: CPU, Memory, I/O limits → resource control OverlayFS: Union filesystem for layer stacking → efficient images Image layers (OverlayFS): Layer 4: COPY app.jar /app/ [your code] ← read-write on top Layer 3: RUN apt-get install ... [added tools] ← read-only Layer 2: eclipse-temurin:21-jre [JRE files] ← read-only, SHARED! Layer 1: ubuntu:22.04 [base OS] ← read-only, SHARED! "SHARED" = multiple images use same layer bytes on disk → efficient pull/storage On container start: thin writable layer added on top Container deletion: only writable layer removed

Observability — Metrics, Logs, Traces

Prometheus, Grafana, ELK, OpenTelemetry, distributed tracing, microservice debugging

1. The Three Pillars

📊 Metrics

Numerical time-series data. CPU%, request rate, error rate, latency percentiles. Aggregated, low storage cost. Answer: "Is the system healthy?"

📝 Logs

Timestamped text records of events. Structured JSON preferred. High detail, high storage cost. Answer: "What exactly happened?"

🔗 Traces

Execution flow across services. Spans with timing, parent-child relationship. Distributed causality. Answer: "Why is this request slow?"

public class OrderService {

    private final Counter orderFailedCounter;
    private final Timer  orderProcessingTimer;
    private final Gauge  activeOrdersGauge;

    public OrderService(MeterRegistry registry) {
        orderCreatedCounter = Counter.builder("orders.created.total")
            .description("Total orders created")
            .tag("app", "order-service")
            .register(registry);

        orderProcessingTimer = Timer.builder("orders.processing.duration")
            .description("Order processing latency")
            .publishPercentiles(0.5, 0.95, 0.99)
            .publishPercentileHistogram()
            .register(registry);

        activeOrdersGauge = Gauge.builder("orders.active",
            orderRepo, OrderRepository::countActive)
            .register(registry);
    }

    public Order createOrder(CreateOrderRequest req) {
        return orderProcessingTimer.record(() -> {
            Order order = doCreate(req);
            orderCreatedCounter.increment();
            return order;
        });
    }
}

@RestController

    @PostMapping("/api/location/update")
    public void updateLocation(
            @RequestHeader("X-Driver-Id") String driverId,
            @RequestBody LocationUpdate update) {

        // Key per city for locality: drivers:geo:NYC

        ));

        // Publish to Kafka for analytics + realtime rider map update
        kafkaTemplate.send("driver-locations", driverId,
            new DriverLocationEvent(driverId, update));
    }

3. Ride Matching Algorithm

@Service
public class RideMatchingService {

    public DriverMatch findDriver(RideRequest request) {
        String geoKey = "drivers:geo:" + request.getCityId();

        // Step 1: Find nearby available drivers (Redis GEO)
        List<GeoResult<RedisGeoCommands.GeoLocation<String>>> nearby =
            redis.opsForGeo().radius(geoKey,
                new Circle(new Point(request.getPickupLng(), request.getPickupLat()),
                    new Distance(5, Metrics.KILOMETERS)),
                GeoRadiusCommandArgs.newGeoRadiusArgs()
                    .includeDistance().sortAscending().limit(20));

        // Step 2: Filter available (not on a trip)
        List<String> availableDriverIds = nearby.stream()
            .map(r -> r.getContent().getName())
            .filter(driverId -> isAvailable(driverId))
            .filter(driverId -> matchesVehicleType(driverId, request.getVehicleType()))
            .limit(5)
            .toList();

        // Step 3: Score and rank (ETA + rating + acceptance rate)
        return availableDriverIds.stream()
            .map(id -> scoreDriver(id, request))
            .max(Comparator.comparingDouble(DriverMatch::getScore))
            .orElseThrow(() -> new NoDriverAvailableException());
    }

    private boolean isAvailable(String driverId) {
        // Check driver not already matched (optimistic locking via Redis)
        String status = (String) redis.opsForHash().get("driver:meta:" + driverId, "status");
        return "AVAILABLE".equals(status);
    }

    public void assignDriver(String rideId, String driverId) {
        // Atomic lock: only ONE rider can claim this driver
        String lockKey = "driver:lock:" + driverId;
        Boolean locked = redis.opsForValue().setIfAbsent(lockKey, rideId, Duration.ofSeconds(30));
        if (!Boolean.TRUE.equals(locked)) {
            throw new DriverAlreadyAssignedException();
        }
        // Update status atomically
        redis.opsForHash().put("driver:meta:" + driverId, "status", "ON_TRIP");
        redis.opsForHash().put("driver:meta:" + driverId, "currentRideId", rideId);
    }
}

2. Calendar Locking — Preventing Double Booking

// Calendar availability stored as date ranges in PostgreSQL
-- availability_blocks table:
-- listing_id, start_date, end_date, type (BOOKED/BLOCKED/AVAILABLE)
-- CONSTRAINT: no overlapping ranges for same listing (exclusion constraint)
CREATE TABLE reservations (
    id           UUID PRIMARY KEY,
    listing_id   UUID NOT NULL,
    check_in     DATE NOT NULL,
    check_out    DATE NOT NULL,
    status       VARCHAR(20),
    EXCLUDE USING gist (
        listing_id WITH =,
        daterange(check_in, check_out) WITH &&
    )  -- PostgreSQL exclusion constraint: no overlapping ranges!
);

// Booking flow with optimistic locking:
@Transactional(isolation = Isolation.READ_COMMITTED)
public BookingResult createBooking(BookingRequest req) {
    String lockKey = "calendar:lock:" + req.getListingId() + ":" + req.getCheckIn() + ":" + req.getCheckOut();

    // Step 1: Acquire distributed lock (prevent concurrent booking attempts)
    String lockToken = UUID.randomUUID().toString();
    if (!redis.opsForValue().setIfAbsent(lockKey, lockToken, Duration.ofSeconds(30))) {
        throw new ConcurrentBookingException("Dates being booked, please retry");
    }

    try {
        // Step 2: Check availability (within distributed lock)
        boolean available = reservationRepo.isAvailable(
            req.getListingId(), req.getCheckIn(), req.getCheckOut());
        if (!available) throw new DatesNotAvailableException();

        // Step 3: Create PENDING reservation
        Reservation reservation = Reservation.create(req);
        reservation.setStatus(PENDING);
        reservationRepo.save(reservation);  // exclusion constraint catches races

        // Step 4: Process payment (outside main tx ideally, but simplified here)
        PaymentResult payment = paymentService.charge(
            req.getGuestId(), req.getTotalAmount(), reservation.getId());

        if (payment.isSuccess()) {
            reservation.confirm(payment.getTransactionId());
            reservationRepo.save(reservation);
            // Publish event → invalidate calendar cache + notify host
            eventPublisher.publish(new BookingConfirmedEvent(reservation));
            return BookingResult.success(reservation);
        } else {
            reservation.fail();
            reservationRepo.save(reservation);
            throw new PaymentFailedException(payment.getErrorMessage());
        }
    } finally {
        releaseLock(lockKey, lockToken);
    }
}

3. Search — Elasticsearch + Redis Cache

// Search query: "San Francisco, 2 guests, Jun 1-5"
@Service
public class ListingSearchService {

    public SearchResult search(SearchRequest req) {
        // Cache key: hash of search params
        String cacheKey = "search:" + DigestUtils.md5Hex(req.toQueryString());

        return Optional.ofNullable((SearchResult) redis.opsForValue().get(cacheKey))
            .orElseGet(() -> {
                SearchResult result = elasticSearch(req);
                // Cache for 60s — availability changes so can't cache long
                redis.opsForValue().set(cacheKey, result, Duration.ofSeconds(60));
                return result;
            });
    }

    private SearchResult elasticSearch(SearchRequest req) {
        // Elasticsearch query:
        // 1. Geo: bounding box around search location
        // 2. Availability: listing NOT in reservations for those dates
        // 3. Capacity: sleeps >= guests
        // 4. Pricing: within budget range
        // 5. Score: relevance + reviews + superhost status
        BoolQueryBuilder query = QueryBuilders.boolQuery()
            .filter(QueryBuilders.geoDistanceQuery("location")
                .point(req.getLat(), req.getLng()).distance("50km"))
            .filter(QueryBuilders.rangeQuery("max_guests").gte(req.getGuests()))
            .filter(QueryBuilders.rangeQuery("price_per_night")
                .gte(req.getMinPrice()).lte(req.getMaxPrice()))
            .mustNot(QueryBuilders.termsQuery("_id", getUnavailableListings(req)));

        return elasticsearchTemplate.search(
            new NativeSearchQueryBuilder().withQuery(query)
                .withPageable(PageRequest.of(req.getPage(), 20)).build(),
            ListingDocument.class);
    }
}

System Design: Twitter Timeline

Fanout-on-write vs fanout-on-read, Kafka, Redis, celebrity problem, hybrid approach

Scale

Traffic

350M users. 500M tweets/day (5,800/sec). 300B timeline reads/day (3.5M/sec). Follows: avg 200, celebrities: 150M.

Key Challenge

A tweet must appear in followers' timelines within seconds. 3.5M read/sec with low latency. Celebrity fan-out: 1 tweet → 150M timeline writes.

Requirement

Timeline read: <50ms. Tweet post: <200ms (post success, not fan-out complete). Eventual timeline consistency acceptable.

1. Two Approaches

Fanout-on-Read (Pull Model)
// On timeline request:
// 1. Get user's 200 follows
// 2. Fetch tweets from each → 200 queries
// 3. Merge, sort by time, paginate

// Pro: simple, no fan-out cost on write
// Con: read is SLOW — 200+ queries
// Con: 3.5M reads/sec × 200 queries = 700M DB queries/sec!
// Unpractical at Twitter scale
Fanout-on-Write (Push Model)
// On tweet post:
// 1. Save tweet to DB
// 2. For each follower: push tweet to their Redis timeline list

// On timeline request:
// 1. LRANGE user:timeline:{userId} 0 50 → 1 Redis query!
// Sub-50ms reads for any user

// Con: tweet by 150M-follower celebrity = 150M Redis writes!
// Katy Perry's tweet → 150M inserts → hours of backlog

2. Hybrid Approach — Twitter's Real Solution

Hybrid: Fanout-on-write for normal users, fanout-on-read for celebrities Define: celebrity = user with > 1M followers (roughly 50K Twitter accounts) Tweet Posted by @normaluser (5K followers): → Save to tweet DB → Kafka: tweet-events → Fan-out workers: push tweetId to 5K follower Redis lists → Done in seconds Tweet Posted by @celebrity (150M followers): → Save to tweet DB → Kafka: tweet-events → NO fan-out! Just mark "new tweet available" Timeline Read for @user: 1. LRANGE user:timeline:{userId} 0 50 ← pre-computed, has normal user tweets 2. For each celebrity @user follows: GET latest tweets from celebrity:{celebId}:tweets (sorted set) Filter: tweetId > user's last-seen-celeb-tweet 3. Merge: combine pre-computed list + live celebrity tweets 4. Sort by timestamp → return top 50 Result: 1 Redis read + ~5 celebrity lookups = 6 reads, still <50ms

3. Implementation

// Tweet entity + storage
@Service
public class TweetService {

    public Tweet postTweet(String userId, String text) {
        // 1. Persist tweet
        Tweet tweet = Tweet.create(userId, text);
        tweetRepo.save(tweet);  // PostgreSQL (sharded by tweetId)

        // 2. Cache in Redis for fast retrieval
        // Sorted set by timestamp score
        redis.opsForZSet().add("user:tweets:" + userId,
            tweet.getId(), tweet.getCreatedAt().toEpochMilli());
        redis.opsForZSet().removeRange("user:tweets:" + userId, 0, -1001);  // keep 1000

        // 3. Fan-out via Kafka
        kafkaTemplate.send("tweet-events", userId, new TweetCreatedEvent(tweet));

        return tweet;
    }
}

// Fan-out worker (Kafka consumer)
@KafkaListener(topics = "tweet-events", groupId = "fanout-workers", concurrency = "20")
public void fanOut(TweetCreatedEvent event) {
    User author = userService.findById(event.getUserId());

    // Skip fan-out for celebrities
    if (author.getFollowerCount() > 1_000_000) {
        // Just update celebrity's tweet list in Redis (for pull-based reads)
        redis.opsForZSet().add("celebrity:tweets:" + event.getUserId(),
            event.getTweetId(), event.getTimestamp());
        return;
    }

    // Get all followers (paginated from DB)
    int page = 0;
    while (true) {
        List<String> followerIds = followRepo.findFollowers(event.getUserId(),
            PageRequest.of(page++, 500));
        if (followerIds.isEmpty()) break;

        // Push to each follower's Redis timeline list
        followerIds.forEach(followerId ->
            redis.opsForList().leftPush("user:timeline:" + followerId, event.getTweetId()));
    }
}

// Timeline read (hybrid merge)
public List<Tweet> getTimeline(String userId) {
    // 1. Pre-computed timeline from Redis
    List<String> tweetIds = redis.opsForList().range("user:timeline:" + userId, 0, 99);

    // 2. Fetch celebrity tweets in parallel
    List<String> celebFollows = followRepo.findCelebrityFollows(userId);  // cached
    List<String> celebTweetIds = celebFollows.parallelStream()
        .flatMap(celebId -> {
            Set<String> ids = redis.opsForZSet().reverseRange("celebrity:tweets:" + celebId, 0, 19);
            return ids.stream();
        })
        .collect(Collectors.toList());

    // 3. Merge and sort
    Set<String> allIds = new LinkedHashSet<>(tweetIds);
    allIds.addAll(celebTweetIds);

    // 4. Batch fetch tweet details (Redis or DB)
    return tweetService.findAllByIds(new ArrayList<>(allIds)).stream()
        .sorted(Comparator.comparing(Tweet::getCreatedAt).reversed())
        .limit(50).toList();
}
