Run all synchronous code in Call Stack 2. Run ALL microtasks (Promises) — may add more microtasks 3. Render (browser repaints DOM) 4. Pick ONE task from Callback Queue (setTimeout, setInterval, I/O) 5. Repeat Key insight: Promises run before setTimeout even if both are "ready"

setTimeout(() => console.log("2"), 0);    // 4 — macro task

2. Virtual DOM & Reconciliation

Render cycle: State/Props change │ ▼ Component function re-executes │ returns React elements (plain objects, cheap!) ▼ New Virtual DOM tree │ React's "diffing" algorithm (reconciliation) ▼ Diff vs Previous Virtual DOM │ Only compute WHAT changed ▼ Minimal DOM updates │ Apply only the diff to real DOM (expensive operations minimized) ▼ Browser repaints Reconciliation rules: 1. Different element type → unmount old, mount new (no reuse!) 2. Same type → update props in place (DOM node reused) 3. Lists without keys → assume position-based (slow, buggy) 4. Lists with stable keys → match by key (fast, correct)

2. useEffect — The Complete Guide

// useEffect(fn, deps)
// deps=undefined → runs after EVERY render
// deps=[]        → runs ONCE after mount
// deps=[a,b]     → runs when a or b changes (shallow comparison)

function UserProfile({ userId }) {
  const [user, setUser] = useState(null);

  useEffect(() => {
    let cancelled = false;                    // cleanup flag

    const loadUser = async () => {
      const data = await fetchUser(userId);
      if (!cancelled) setUser(data);           // don't update unmounted component
    };

    loadUser();

    return () => { cancelled = true; };       // cleanup on userId change/unmount
  }, [userId]);                                // re-run when userId changes

  // AbortController is cleaner:
  useEffect(() => {
    const controller = new AbortController();
    fetch(`/api/users/${userId}`, { signal: controller.signal })
      .then(r => r.json()).then(setUser)
      .catch(e => { if (e.name !== "AbortError") console.error(e); });
    return () => controller.abort();
  }, [userId]);
}

3. Stale Closure — The #1 Hook Bug

❌ Stale closure — reads old count
function Timer() {
  const [count, setCount] = useState(0);

  useEffect(() => {
    const id = setInterval(() => {
      // 'count' captured at effect creation!
      // Always logs 0, never increments
      console.log(count); // stale!
      setCount(count + 1); // always sets to 1!
    }, 1000);
    return () => clearInterval(id);
  }, []);  // empty deps = closure captures count=0
}
✅ Functional update fixes stale closure
function Timer() {
  const [count, setCount] = useState(0);

  useEffect(() => {
    const id = setInterval(() => {
      // Functional update: always gets CURRENT count
      setCount(prev => prev + 1); // ✅
    }, 1000);
    return () => clearInterval(id);
  }, []);  // no deps needed!

  return <div>{count}</div>;
}

4. useMemo & useCallback — When to Use

// useMemo: memoize expensive computation
function ExpensiveList({ items, filter }) {
  // Recalculates ONLY when items or filter changes
  const filteredItems = useMemo(
    () => items.filter(item => item.category === filter).sort((a, b) => a.price - b.price),
    [items, filter]  // dependencies
  );
  return filteredItems.map(item => <Item key={item.id} item={item} />);
}

// useCallback: stable function reference for children/effects
function Parent() {
  const [count, setCount] = useState(0);

  // Without useCallback: new function every render → Child re-renders every time
  // With useCallback: same reference if count hasn't changed
  const handleSubmit = useCallback((data) => {
    submitForm(data, count);
  }, [count]);  // depends on count

  return <MemoizedChild onSubmit={handleSubmit} />;
}

// React.memo: skip re-render if props unchanged
const MemoizedChild = React.memo(function Child({ onSubmit }) {
  console.log("Child renders");
  return <button onClick={onSubmit}>Submit</button>;
});

// ⚠️ Don't over-memoize!
// Memoization has overhead. Only use when:

5. Custom Hooks — Reusable Logic

// Custom hooks: extract and reuse stateful logic
function useLocalStorage<T>(key: string, defaultValue: T) {
  const [value, setValue] = useState<T>(() => {
    try {
      const stored = localStorage.getItem(key);
      return stored ? JSON.parse(stored) : defaultValue;
    } catch { return defaultValue; }
  });

  const setStored = useCallback((newValue: T | ((v: T) => T)) => {
    setValue(prev => {
      const next = typeof newValue === "function"
        ? (newValue as (v: T) => T)(prev) : newValue;
      localStorage.setItem(key, JSON.stringify(next));
      return next;
    });
  }, [key]);

  return [value, setStored] as const;
}

// useDebounce hook
function useDebounce<T>(value: T, delay = 300): T {
  const [debounced, setDebounced] = useState(value);
  useEffect(() => {
    const timer = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(timer);
  }, [value, delay]);
  return debounced;
}

// Usage:
const [search, setSearch] = useState("");
const debouncedSearch = useDebounce(search, 400);
useEffect(() => { if (debouncedSearch) fetchResults(debouncedSearch); }, [debouncedSearch]);

2. State Colocation Rule

Rule: State lives as close to where it's used as possible Local state: used only in one component → useState inside that component Shared state: used by siblings → lift to closest common ancestor Global state: used everywhere → Context or Redux Accordion.expanded → local useState (only Accordion needs it) Modal.isOpen → lift to parent (parent controls open/close) ProductList.filter → URL search params (shareable, bookmarkable) User.authToken → global Redux/Context (needed everywhere) Anti-pattern: lifting state too high Putting everything in Redux/Context when it's only used locally → Unnecessary re-renders of unrelated components → Harder to reason about, harder to test in isolation

// ProtectedRoute: redirect if not authenticated
function ProtectedRoute({ children }: { children: React.ReactNode }) {
  const { isAuthenticated, isLoading } = useAuth();
  const location = useLocation();

  if (isLoading) return <PageSpinner />;

  if (!isAuthenticated) {

1. Axios Client Setup

// shared/api/client.ts
import axios, { AxiosError } from "axios";
import { tokenService } from "../services/tokenService";
import { refreshTokens } from "../../features/auth/api/authApi";

export const apiClient = axios.create({
  baseURL: import.meta.env.VITE_API_URL ?? "http://localhost:8080",
  timeout: 15000,
  headers: { "Content-Type": "application/json" },
});

// Request interceptor: attach JWT
apiClient.interceptors.request.use((config) => {
  const token = tokenService.getAccessToken();
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

// Response interceptor: handle 401 → refresh token
let isRefreshing = false;
let refreshQueue: Array<(token: string) => void> = [];

apiClient.interceptors.response.use(
  (res) => res,
  async (error: AxiosError) => {
    const original = error.config as any;
    if (error.response?.status !== 401 || original._retry) return Promise.reject(error);
    original._retry = true;

    if (isRefreshing) {
      // Queue requests while refreshing
      return new Promise((resolve) => {
        refreshQueue.push((token) => {
          original.headers.Authorization = `Bearer ${token}`;
          resolve(apiClient(original));
        });
      });
    }

    isRefreshing = true;
    try {
      const { accessToken } = await refreshTokens();
      tokenService.setTokens(accessToken);
      refreshQueue.forEach(cb => cb(accessToken));
      refreshQueue = [];
      original.headers.Authorization = `Bearer ${accessToken}`;
      return apiClient(original);
    } catch (refreshError) {
      tokenService.clear();
      window.location.href = "/login";
      return Promise.reject(refreshError);
    } finally {
      isRefreshing = false;
    }
  }
);

2. TanStack Query (React Query) — The Gold Standard

// features/orders/hooks/useOrders.ts
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";

// Fetching with automatic caching, refetch, loading states
export function useOrders(filters: OrderFilters) {
  return useQuery({
    queryKey: ["orders", filters],       // cache key (array)
    queryFn: () => orderApi.getOrders(filters),
    staleTime: 5 * 60 * 1000,             // 5 min: don't refetch if fresh
    retry: 2,                               // retry failed requests twice
    retryDelay: (attempt) => Math.min(1000 * 2 ** attempt, 10000),
  });
}

// Mutation with optimistic update
export function useUpdateOrderStatus() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: ({ id, status }: { id: string, status: string }) =>
      orderApi.updateStatus(id, status),

    // Optimistic update: update UI BEFORE server confirms
    onMutate: async ({ id, status }) => {
      await queryClient.cancelQueries({ queryKey: ["orders"] });
      const snapshot = queryClient.getQueryData(["orders"]);
      queryClient.setQueryData(["orders"], (old: Order[]) =>
        old.map(o => o.id === id ? { ...o, status } : o)
      );
      return { snapshot };  // context for rollback
    },
    onError: (err, vars, ctx) => {
      // Rollback to snapshot on error
      queryClient.setQueryData(["orders"], ctx?.snapshot);
    },
    onSettled: () => {
      // Always refetch to sync with server truth
      queryClient.invalidateQueries({ queryKey: ["orders"] });
    },
  });
}

// Usage in component:
function OrdersPage() {
  const [filters, setFilters] = useState({ status: "all", page: 1 });
  const { data: orders, isLoading, isError, error } = useOrders(filters);

  if (isLoading) return <TableSkeleton rows={10} />;
  if (isError)   return <ErrorAlert message={error.message} />;

  return (
    <OrderTable

function InfiniteOrderList() {
  const { data, fetchNextPage, hasNextPage, isFetchingNextPage } = useInfiniteOrders();
  const sentinelRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => { if (entry.isIntersecting && hasNextPage) fetchNextPage(); },
      { threshold: 0.1 }
    );
    if (sentinelRef.current) observer.observe(sentinelRef.current);
    return () => observer.disconnect();

  return (
    <div>
      {data?.pages.flatMap(page => page.orders).map(order => (
        <OrderRow key={order.id} order={order} />
      ))}
      <div ref={sentinelRef}>{isFetchingNextPage && <Spinner />}</div>

// Context: good for slowly-changing global data (theme, auth, locale)
// NOT for high-frequency updates (use Redux/Zustand)

type AuthContextType = {
  user: User | null;
  isAuthenticated: boolean;
  login: (credentials: Credentials) => Promise<void>;
  logout: () => void;
};

const AuthContext = React.createContext<AuthContextType>(null!);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(() =>
    tokenService.getUser()     // lazy init from localStorage
  );

  const login = useCallback(async (credentials) => {
    const { user, tokens } = await authApi.login(credentials);
    tokenService.setTokens(tokens);
    setUser(user);
  }, []);

  const logout = useCallback(() => {
    tokenService.clear();
    setUser(null);
  }, []);

  // Memoize value to prevent unnecessary re-renders
  const value = useMemo(() =>
    ({ user, isAuthenticated: !!user, login, logout }),

  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

1. Token Service

// shared/services/tokenService.ts
// NEVER store tokens in localStorage (XSS risk for refresh tokens)
// Best: httpOnly cookie (backend) + short-lived access token in memory
// Pragmatic: access token in memory, refresh token in localStorage

const TOKEN_KEY = "app_access_token";
const REFRESH_KEY = "app_refresh_token";

let inMemoryToken: string | null = null;

export const tokenService = {
  getAccessToken: () => inMemoryToken,
  getRefreshToken: () => localStorage.getItem(REFRESH_KEY),

  setTokens: ({ accessToken, refreshToken }: Tokens) => {
    inMemoryToken = accessToken;                               // memory only
    if (refreshToken) localStorage.setItem(REFRESH_KEY, refreshToken);
  },

  getUser: (): User | null => {
    const token = inMemoryToken ?? localStorage.getItem(TOKEN_KEY);
    if (!token) return null;
    try {
      const payload = JSON.parse(atob(token.split(".")[1]));
      if (payload.exp * 1000 < Date.now()) return null;  // expired
      return { id: payload.sub, email: payload.email, roles: payload.roles };
    } catch { return null; }
  },

  isTokenExpiringSoon: () => {
    const token = inMemoryToken;
    if (!token) return true;
    const { exp } = JSON.parse(atob(token.split(".")[1]));
    return (exp * 1000 - Date.now()) < 60 * 1000;  // within 1 minute
  },

  clear: () => {
    inMemoryToken = null;
    localStorage.removeItem(REFRESH_KEY);
  },
};

2. Silent Refresh — Proactive Token Renewal

// Silently refresh token before expiry
function useTokenRefresh() {
  const { logout } = useAuth();

  useEffect(() => {
    const scheduleRefresh = () => {
      const token = tokenService.getAccessToken();
      if (!token) return;

      const { exp } = JSON.parse(atob(token.split(".")[1]));
      const msUntilExpiry = exp * 1000 - Date.now();
      const refreshIn = msUntilExpiry - 60 * 1000;  // 1 min early

      if (refreshIn <= 0) { logout(); return; }

      const timer = setTimeout(async () => {
        try {
          const { accessToken } = await authApi.refreshToken(tokenService.getRefreshToken()!);
          tokenService.setTokens({ accessToken });
          scheduleRefresh();  // schedule next refresh
        } catch { logout(); }
      }, refreshIn);

      return () => clearTimeout(timer);
    };
    return scheduleRefresh();
  }, [logout]);
}

    }).then(r => r.data);
  },

function FileUpload({ orderId }: { orderId: string }) {
  const [progress, setProgress] = useState(0);
  const [status, setStatus] = useState<"idle"|"uploading"|"done"|"error">("idle");
  const queryClient = useQueryClient();

  const handleChange = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;

    // Validate file
    if (file.size > 10 * 1024 * 1024) return alert("Max 10MB");
    if (!["image/jpeg", "image/png", "application/pdf"].includes(file.type)) {
      return alert("Only JPG, PNG, PDF");
    }

    setStatus("uploading"); setProgress(0);
    try {
      await orderApi.uploadAttachment(orderId, file, setProgress);
      setStatus("done");
      queryClient.invalidateQueries({ queryKey: ["orders", orderId] });
    } catch { setStatus("error"); }
  };

  return (
    <div className="upload-zone">
      <input type="file" onChange={handleChange} accept=".jpg,.png,.pdf" />
      {status === "uploading" && <ProgressBar value={progress} />}
      {status === "done"      && <span className="text-green-500">✓ Uploaded</span>}
      {status === "error"     && <span className="text-red-500">✗ Failed</span>}
    </div>
  );
}
